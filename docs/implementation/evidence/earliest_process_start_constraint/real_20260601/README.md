```bash
python tools/verify_earliest_start_real_data.py prepare \
  --csv '/Users/miles/Desktop/GQGA4-4-计算最晚日期-2026-06 (2).csv' \
  --start '2026-06-01T00:00:00+08:00' --output /tmp/apsgo-earliest-real-20260917
```
